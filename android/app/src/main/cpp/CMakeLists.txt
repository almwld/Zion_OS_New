cmake_minimum_required(VERSION 3.22.1)
project(zion_native LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

add_library(zionpty SHARED zion_pty.c)
find_library(log-lib log)
target_link_libraries(zionpty ${log-lib})

if(ANDROID_ABI STREQUAL "arm64-v8a")
  include(FetchContent)
  FetchContent_Declare(llama_cpp URL https://github.com/ggml-org/llama.cpp/archive/refs/tags/b11193.tar.gz DOWNLOAD_EXTRACT_TIMESTAMP TRUE)
  set(LLAMA_BUILD_TESTS OFF CACHE BOOL "" FORCE)
  set(LLAMA_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
  set(LLAMA_BUILD_SERVER OFF CACHE BOOL "" FORCE)
  set(LLAMA_BUILD_COMMON OFF CACHE BOOL "" FORCE)
  set(LLAMA_OPENSSL OFF CACHE BOOL "" FORCE)
  set(LLAMA_CURL OFF CACHE BOOL "" FORCE)
  set(LLAMA_HTTPLIB OFF CACHE BOOL "" FORCE)
  set(GGML_OPENMP OFF CACHE BOOL "" FORCE)
  set(GGML_LLAMAFILE OFF CACHE BOOL "" FORCE)
  set(GGML_NATIVE OFF CACHE BOOL "" FORCE)
  set(GGML_CPU_ALL_VARIANTS ON CACHE BOOL "" FORCE)
  set(GGML_OPENCL OFF CACHE BOOL "" FORCE)
  FetchContent_MakeAvailable(llama_cpp)
  add_library(llama_jni SHARED llama_jni.cpp)
  target_link_libraries(llama_jni llama ${log-lib})
  target_compile_options(llama_jni PRIVATE -ffunction-sections -fdata-sections -fvisibility=hidden)
else()
  add_library(llama_jni SHARED llama_jni_stub.cpp)
  target_link_libraries(llama_jni ${log-lib})
endif()
